```cpp
namespace helengine::n64 {
/// Maximum directional lights Tiny3D can evaluate in one draw.
constexpr int kMaxDirectionalLights = 7;

/// One directional light converted to Tiny3D's 8-bit color and surface-to-light unit vector.
struct Nintendo64DirectionalLight {
    uint8_t Color[4];
    float Direction[3];
};

/// Ambient color plus up to seven directional lights for one rendered frame.
struct Nintendo64FrameLights {
    uint8_t Ambient[4] = {0, 0, 0, 255};
    Nintendo64DirectionalLight Directional[kMaxDirectionalLights] = {};
    int DirectionalCount = 0;
};

class Nintendo64LightMath {
public:
    /// Scales an RGB color by intensity, clamps each channel to [0,1], and packs it as RGBA8 with alpha 255.
    static void PackColor(float red, float green, float blue, float intensity, uint8_t output[4]);
    /// Adds an intensity-scaled color into a running RGB accumulator (no clamping until PackAccumulated).
    static void Accumulate(float red, float green, float blue, float intensity, float accumulator[3]);
    /// Clamps an accumulated RGB triple to [0,1] and packs it as RGBA8 with alpha 255.
    static void PackAccumulated(const float accumulator[3], uint8_t output[4]);
    /// Converts an entity forward direction into Tiny3D's surface-to-light unit vector (negated and normalized).
    /// Returns false when the input has zero length.
    static bool ToLightVector(float forwardX, float forwardY, float forwardZ, float output[3]);
};

/// Fills Nintendo64FrameLights from the generated ObjectManager light lists.
class Nintendo64LightExtractor {
public:
    /// Throws InvalidOperationException when more than seven directional lights are registered or a light direction is degenerate.
    static void Extract(::ObjectManager* objectManager, Nintendo64FrameLights& lights);
};
}
```

```cpp
/// Clamps one linear channel to [0,1] and rounds it to an 8-bit value.
uint8_t ToByte(float channel) {
    const float clamped = channel < 0.0f ? 0.0f : (channel > 1.0f ? 1.0f : channel);
    return static_cast<uint8_t>(std::lround(clamped * 255.0f));
}

} // namespace

void Nintendo64LightMath::PackColor(float red, float green, float blue, float intensity, uint8_t output[4]) {
    output[0] = ToByte(red * intensity);
    output[1] = ToByte(green * intensity);
    output[2] = ToByte(blue * intensity);
    output[3] = 255;
}

void Nintendo64LightMath::Accumulate(float red, float green, float blue, float intensity, float accumulator[3]) {
    accumulator[0] += red * intensity;
    accumulator[1] += green * intensity;
    accumulator[2] += blue * intensity;
}

void Nintendo64LightMath::PackAccumulated(const float accumulator[3], uint8_t output[4]) {
    output[0] = ToByte(accumulator[0]);
    output[1] = ToByte(accumulator[1]);
    output[2] = ToByte(accumulator[2]);
    output[3] = 255;
}

bool Nintendo64LightMath::ToLightVector(float forwardX, float forwardY, float forwardZ, float output[3]) {
    const float lengthSquared = forwardX * forwardX + forwardY * forwardY + forwardZ * forwardZ;
    if (!(lengthSquared > 0.0f) || !std::isfinite(lengthSquared)) {
        return false;
    }
    const float inverseLength = 1.0f / std::sqrt(lengthSquared);
    output[0] = -forwardX * inverseLength;
    output[1] = -forwardY * inverseLength;
    output[2] = -forwardZ * inverseLength;
    return true;
}

} // namespace helengine::n64
```

Note: `ToByte` is a file-local function in an anonymous namespace, matching the existing `SetPackedVertex` pattern in this repo; it is not a nested local function. `Nintendo64FrameLights.hpp` contains the two structs and `kMaxDirectionalLights` from Interfaces, with `#include <cstdint>` only.

```cpp
void Nintendo64LightExtractor::Extract(::ObjectManager* objectManager, Nintendo64FrameLights& lights) {
    if (objectManager == nullptr) {
        throw new ArgumentNullException("objectManager");
    }

    float ambientAccumulator[3] = {0.0f, 0.0f, 0.0f};
    ::List<::AmbientLightComponent*>* ambientLights = objectManager->get_AmbientLights();
    if (ambientLights != nullptr) {
        for (int32_t index = 0; index < ambientLights->get_Count(); index++) {
            ::AmbientLightComponent* ambient = (*ambientLights)[index];
            if (ambient == nullptr) {
                continue;
            }
            const ::float4 color = ambient->get_Color();
            Nintendo64LightMath::Accumulate(color.X, color.Y, color.Z, ambient->get_Intensity(), ambientAccumulator);
        }
    }
    Nintendo64LightMath::PackAccumulated(ambientAccumulator, lights.Ambient);

    lights.DirectionalCount = 0;
    ::List<::DirectionalLightComponent*>* directionalLights = objectManager->get_DirectionalLights();
    if (directionalLights == nullptr) {
        return;
    }
    if (directionalLights->get_Count() > kMaxDirectionalLights) {
        throw new InvalidOperationException("N64 lighting supports at most seven directional lights.");
    }
    for (int32_t index = 0; index < directionalLights->get_Count(); index++) {
        ::DirectionalLightComponent* directional = (*directionalLights)[index];
        if (directional == nullptr) {
            continue;
        }
        Nintendo64DirectionalLight& entry = lights.Directional[lights.DirectionalCount];
        const ::float4 color = directional->get_Color();
        Nintendo64LightMath::PackColor(color.X, color.Y, color.Z, directional->get_Intensity(), entry.Color);
        const ::float3 forward = ::LightDirectionUtility::GetLightDirection(directional);
        if (!Nintendo64LightMath::ToLightVector(forward.X, forward.Y, forward.Z, entry.Direction)) {
            throw new InvalidOperationException("N64 directional light has a zero-length direction.");
        }
        lights.DirectionalCount++;
    }
}
```
