```csharp
using System.Text.Json;
using helengine.editor;

namespace helengine.n64.builder;

/// <summary>
/// Serializes and parses the N64 texture cook settings payload carried by builder-owned platform cook work items.
/// </summary>
public static class Nintendo64TextureCookSettingsSerializer {
    /// <summary>
    /// Serializes one N64 texture settings payload into the stable string form stored on cook work items.
    /// </summary>
    /// <param name="settings">Settings payload to serialize.</param>
    /// <returns>Serialized N64 texture settings string.</returns>
    public static string Serialize(TextureAssetProcessorSettings settings) {
        if (settings == null) {
            throw new ArgumentNullException(nameof(settings));
        }

        return JsonSerializer.Serialize(new Dictionary<string, object> {
            ["maxResolution"] = settings.MaxResolution,
            ["colorFormat"] = settings.ColorFormat.ToString(),
            ["alphaPrecision"] = settings.AlphaPrecision.ToString()
        });
    }

    /// <summary>
    /// Parses one N64 texture settings payload from the string form stored on cook work items.
    /// </summary>
    /// <param name="serializedSettings">Serialized settings string.</param>
    /// <returns>Parsed N64 texture settings payload.</returns>
    public static TextureAssetProcessorSettings Deserialize(string serializedSettings) {
        if (string.IsNullOrWhiteSpace(serializedSettings)) {
            throw new ArgumentException("Serialized N64 texture settings are required.", nameof(serializedSettings));
        }

        using JsonDocument document = JsonDocument.Parse(serializedSettings);
        JsonElement root = document.RootElement;
        int maxResolution = root.TryGetProperty("maxResolution", out JsonElement maxResolutionElement)
            ? maxResolutionElement.GetInt32()
            : 0;
        string colorFormatName = root.TryGetProperty("colorFormat", out JsonElement colorFormatElement)
            ? colorFormatElement.GetString() ?? TextureAssetColorFormat.Rgba32.ToString()
            : TextureAssetColorFormat.Rgba32.ToString();
        string alphaPrecisionName = root.TryGetProperty("alphaPrecision", out JsonElement alphaPrecisionElement)
            ? alphaPrecisionElement.GetString() ?? TextureAssetAlphaPrecision.A8.ToString()
            : TextureAssetAlphaPrecision.A8.ToString();

        if (!Enum.TryParse(colorFormatName, true, out TextureAssetColorFormat colorFormat)) {
            throw new InvalidOperationException($"Unsupported N64 texture color format '{colorFormatName}'.");
        }
        if (colorFormat != TextureAssetColorFormat.Rgba32) {
            throw new InvalidOperationException($"N64 textures must cook to Rgba32; received '{colorFormatName}'.");
        }
        if (!Enum.TryParse(alphaPrecisionName, true, out TextureAssetAlphaPrecision alphaPrecision)) {
            throw new InvalidOperationException($"Unsupported N64 texture alpha precision '{alphaPrecisionName}'.");
        }

        return new TextureAssetProcessorSettings {
            MaxResolution = maxResolution,
            ColorFormat = colorFormat,
            AlphaPrecision = alphaPrecision
        };
    }
}
```

```csharp
using System.Drawing;

namespace helengine.n64.builder;

/// <summary>
/// Decodes one authored source image into an uncooked straight-RGBA texture asset for the N64 cook.
/// </summary>
public sealed class Nintendo64SourceTextureDecoder {
    /// <summary>
    /// Decodes one source image file into a raw RGBA texture asset.
    /// </summary>
    /// <param name="sourceAssetPath">Absolute source image path.</param>
    /// <returns>Raw texture asset containing straight RGBA pixel bytes.</returns>
    public TextureAsset Decode(string sourceAssetPath) {
        if (string.IsNullOrWhiteSpace(sourceAssetPath)) {
            throw new ArgumentException("Source asset path is required.", nameof(sourceAssetPath));
        }
        if (!File.Exists(sourceAssetPath)) {
            throw new FileNotFoundException("Source texture file was not found.", sourceAssetPath);
        }

        using Bitmap bitmap = new(sourceAssetPath);
        if (bitmap.Width < 1 || bitmap.Height < 1) {
            throw new InvalidOperationException("Source textures must have positive dimensions.");
        }

        byte[] rgbaBytes = new byte[bitmap.Width * bitmap.Height * 4];
        int writeOffset = 0;
        for (int y = 0; y < bitmap.Height; y++) {
            for (int x = 0; x < bitmap.Width; x++) {
                Color pixel = bitmap.GetPixel(x, y);
                rgbaBytes[writeOffset++] = pixel.R;
                rgbaBytes[writeOffset++] = pixel.G;
                rgbaBytes[writeOffset++] = pixel.B;
                rgbaBytes[writeOffset++] = pixel.A;
            }
        }

        return new TextureAsset {
            Width = checked((ushort)bitmap.Width),
            Height = checked((ushort)bitmap.Height),
            Colors = rgbaBytes,
            PaletteColors = Array.Empty<byte>(),
            ColorFormat = TextureAssetColorFormat.Rgba32,
            AlphaPrecision = TextureAssetAlphaPrecision.A8
        };
    }
}
```

- [ ] **Step 5: Implement the cooker**

Create `builder/Nintendo64TextureCooker.cs` with a `Cook(TextureAsset sourceTexture, TextureAssetProcessorSettings settings)` method that:
1. Throws `ArgumentNullException` for a null texture or settings, and `InvalidOperationException` when `sourceTexture.ColorFormat != Rgba32` or `Colors` is null or the wrong length for `Width * Height * 4`.
2. Computes `int maxSide = settings.MaxResolution > 0 ? settings.MaxResolution : int.MaxValue;` and the target size: if both `Width` and `Height` are already `<= maxSide`, returns a copy with the same pixels; otherwise scales both by `double factor = maxSide / (double)Math.Max(Width, Height)` and rounds each side with `Math.Max(1, (int)Math.Round(side * factor))`.
3. Box-filters: for each destination pixel, averages the source pixels in the rectangle `[x * Width / targetWidth, (x + 1) * Width / targetWidth)` by `[y * Height / targetHeight, (y + 1) * Height / targetHeight)`, accumulating R, G, B, A as integers and dividing by the sample count. Use at least one sample per destination pixel.
4. When `settings.AlphaPrecision == TextureAssetAlphaPrecision.Opaque`, writes 255 into every alpha byte after filtering.
5. Returns a new `TextureAsset` with the target dimensions, `ColorFormat = Rgba32`, `AlphaPrecision = settings.AlphaPrecision`, `PaletteColors = Array.Empty<byte>()`, and the filtered bytes. It does not set `Id`; the executor does.

Every method and the class get substantive XML doc comments. No local functions: if a private helper is needed, make it a private method on the class.

using helengine.baseplatform.Manifest;
using helengine.editor;

namespace helengine.n64.builder;

/// <summary>
/// Executes builder-owned N64 platform cook work items by writing cooked runtime payloads into the package source root.
/// </summary>
public sealed class Nintendo64PlatformCookWorkItemExecutor {
    /// <summary>
    /// Decodes and cooks one source texture work item into the package source root.
    /// </summary>
    /// <param name="workItem">Work item describing the source asset, settings, and cooked output path.</param>
    /// <param name="packageSourceRootPath">Package source root that the DragonFS stager reads.</param>
    public void Execute(PlatformCookWorkItem workItem, string packageSourceRootPath) {
        if (workItem == null) {
            throw new ArgumentNullException(nameof(workItem));
        }
        if (string.IsNullOrWhiteSpace(packageSourceRootPath)) {
            throw new ArgumentException("Package source root path must be provided.", nameof(packageSourceRootPath));
        }
        if (!string.Equals(workItem.SourceAssetKind, "texture", StringComparison.OrdinalIgnoreCase)) {
            throw new InvalidOperationException($"N64 does not support source asset kind '{workItem.SourceAssetKind}'.");
        }

        TextureAssetProcessorSettings settings = Nintendo64TextureCookSettingsSerializer.Deserialize(workItem.SerializedPlatformSettings);
        TextureAsset cookedTexture = TextureCooker.Cook(SourceTextureDecoder.Decode(workItem.SourceAssetPath), settings);
        cookedTexture.Id = workItem.OutputLogicalArtifactId;

        string destinationPath = Path.Combine(
            Path.GetFullPath(packageSourceRootPath),
            Nintendo64ContentStager.NormalizeRelativePath(workItem.OutputRelativePath).Replace('/', Path.DirectorySeparatorChar));
        string destinationDirectory = Path.GetDirectoryName(destinationPath)
            ?? throw new InvalidOperationException($"N64 cooked texture destination directory could not be resolved for '{destinationPath}'.");
        Directory.CreateDirectory(destinationDirectory);
        File.WriteAllBytes(destinationPath, helengine.files.AssetSerializer.SerializeToBytes(cookedTexture));
    }

    /// <summary>
    /// Decodes authored source images into raw texture assets.
    /// </summary>
    readonly Nintendo64SourceTextureDecoder SourceTextureDecoder = new();

    /// <summary>
    /// Downscales decoded textures into the cooked N64 runtime form.
    /// </summary>
    readonly Nintendo64TextureCooker TextureCooker = new();
}
```

uint16_t Nintendo64TexelPacking::PackRgba5551(uint8_t red, uint8_t green, uint8_t blue, uint8_t alpha) {
    const uint16_t packedRed = static_cast<uint16_t>((red >> 3) & 0x1Fu);
    const uint16_t packedGreen = static_cast<uint16_t>((green >> 3) & 0x1Fu);
    const uint16_t packedBlue = static_cast<uint16_t>((blue >> 3) & 0x1Fu);
    const uint16_t packedAlpha = static_cast<uint16_t>(alpha >= 128u ? 1u : 0u);
    return static_cast<uint16_t>((packedRed << 11) | (packedGreen << 6) | (packedBlue << 1) | packedAlpha);
}

void Nintendo64TexelPacking::ConvertRow(const uint8_t* rgba, int32_t width, uint16_t* output) {
    if (rgba == nullptr || output == nullptr || width <= 0) {
        return;
    }
    for (int32_t index = 0; index < width; index++) {
        const uint8_t* texel = rgba + static_cast<size_t>(index) * 4u;
        output[index] = PackRgba5551(texel[0], texel[1], texel[2], texel[3]);
    }
}
```
